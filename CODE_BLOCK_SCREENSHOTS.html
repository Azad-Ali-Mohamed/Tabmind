2. Adaptive retention score

versions/v1.5.0/background.js — score()
const idle = Math.min((now - item.lastAccess) / (6 * HOUR), 1);
const aging = 1 - idle;
const recency = Math.exp(-0.7 * (now - item.lastAccess) / HOUR);
const frequency = Math.min(
  Math.log1p(item.visits) / Math.log1p(MAX_VISITS), 1
) * aging;
const activeTime = Math.min(item.activeMs / 120000, 1) * aging;
const priority = item.groupId !== -1 ? 0.8 : 0.35;
const focus = active && tab.windowId === active.windowId ? 0.7 : 0;
const cluster = clusterScore(item, active);

const retention =
  0.35 * recency +
  0.20 * frequency +
  0.15 * activeTime +
  0.10 * priority +
  0.10 * focus +
  0.10 * cluster;

3. Protection and eviction

Hard protection plus lowest-score selection
function protectedTab(tab, domains) {
  const h = host(tab);
  return tab.active || tab.pinned ||
    domains.some(d => h === d || h.endsWith("." + d));
}

const candidates = open.filter(
  tab => !protectedTab(tab, domains)
);
candidates.sort((a, b) =>
  score(a, active).retention - score(b, active).retention
);
if (candidates[0]) {
  await chrome.tabs.discard(candidates[0].id);
}

4. New-tab protection notification

Wait for a real HTTP/HTTPS domain
function host(tab) {
  try {
    const url = new URL(tab.url || "");
    return ["http:", "https:"].includes(url.protocol)
      ? url.hostname.toLowerCase()
      : "";
  } catch { return ""; }
}

chrome.tabs.onCreated.addListener(tab => {
  entry(tab);
  pendingPrompts.add(tab.id);
});

chrome.tabs.onUpdated.addListener((tabId, changeInfo, tab) => {
  if (pendingPrompts.has(tabId) &&
      (changeInfo.url || changeInfo.status === "complete")) {
    notifyNewDomain(tab);
  }
});

const protect = !safe && h
  ? `<button class="protect-tab"
       data-protect-tab="${esc(h)}"
       aria-label="Protect ${esc(h)}">+</button>`
  : "";

async function add(value) {
  value = value.trim().toLowerCase()
    .replace(/^https?:\/\//, "").split("/")[0];
  if (value && !domains.includes(value)) {
    domains.push(value);
    await save();
  }
}
